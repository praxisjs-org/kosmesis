## Usage

```tsx
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";

@Component()
class AccountSettings extends StatefulComponent {
  @State() tabs = new Tabs({ defaultValue: "account" });

  render() {
    return (
      <Tabs defaultValue="account" class="w-96">
        <TabsList tabs={this.tabs} aria-label="Settings">
          <TabsTrigger tabs={this.tabs} value="account">Account</TabsTrigger>
          <TabsTrigger tabs={this.tabs} value="password">Password</TabsTrigger>
        </TabsList>
        <TabsContent tabs={this.tabs} value="account">Account settings.</TabsContent>
        <TabsContent tabs={this.tabs} value="password">Password settings.</TabsContent>
      </Tabs>
    );
  }
}
```

<Callout type="info">
  `<Tabs>` mounted in the tree and `this.tabs` are two separate objects, kept in sync via matching
  props (`defaultValue`) — see the same note on [Radio Group](/docs/components/radio-group) for
  why. `TabsList`, `TabsTrigger`, and `TabsContent` all read/drive selection through `this.tabs`.
</Callout>

## Props

### Tabs

| Prop | Type | Default |
|---|---|---|
| `value` | `string` | — |
| `defaultValue` | `string` | — |
| `orientation` | `"horizontal" \| "vertical"` | `"horizontal"` |
| `onValueChange` | `(value: string) => void` | — |

### TabsList

| Prop | Type | Default |
|---|---|---|
| `tabs` | `Tabs` | — required |
| `aria-label` | `string` | — |
| `aria-labelledby` | `string` | — |

### TabsTrigger

| Prop | Type | Default |
|---|---|---|
| `tabs` | `Tabs` | — required |
| `value` | `string` | — required |
| `disabled` | `boolean` | — |

### TabsContent

| Prop | Type | Default |
|---|---|---|
| `tabs` | `Tabs` | — required |
| `value` | `string` | — required |
