## About

Purely presentational — no Morphos equivalent. Same runtime `cols`/`colSpan`/`rowSpan`-via-inline-style
approach as `Grid`, styled as bordered cards (`bg-card`) instead of bare tracks.

## Usage

```tsx
import { BentoGrid, BentoGridItem } from "@/components/ui/bento-grid";

<BentoGrid cols={3}>
  <BentoGridItem colSpan={2}>
    <h3>Analytics</h3>
    <p>Track everything in one place.</p>
  </BentoGridItem>
  <BentoGridItem>
    <h3>Team</h3>
  </BentoGridItem>
</BentoGrid>
```
