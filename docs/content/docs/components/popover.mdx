## Usage

```tsx
import { Popover, PopoverContent, PopoverTrigger } from "@morphos/overlays";
import { buttonVariants } from "@/components/ui/button";

@Component()
class Settings extends StatefulComponent {
  @State() popover = new Popover();

  render() {
    return (
      <>
        <PopoverTrigger popover={this.popover} class={buttonVariants({ variant: "outline" })}>
          Open popover
        </PopoverTrigger>
        <PopoverContent popover={this.popover}>
          <p>Place any content here.</p>
        </PopoverContent>
      </>
    );
  }
}
```

<Callout type="warning">
  Morphos's `PopoverTrigger` always renders its own `<button>` — there's no `asChild` merge like
  Radix's. Style it directly with `buttonVariants()` (imported from `./button`) instead of
  nesting a separate `Button` inside it, which would produce invalid nested `<button>` elements.
</Callout>

## Props

### Popover

Re-exported directly from `@morphos/overlays`.

| Prop | Type | Default |
|---|---|---|
| `open` | `boolean` | — controlled open state |
| `defaultOpen` | `boolean` | — |
| `onOpenChange` | `(open: boolean) => void` | — |
| `closeOnEscape` | `boolean` | — |
| `closeOnOutsideClick` | `boolean` | — |
| `side` | `"top" \| "right" \| "bottom" \| "left"` | `"bottom"` |
| `align` | `"start" \| "center" \| "end"` | `"start"` |
| `sideOffset` | `number` | `4` |

### PopoverTrigger

| Prop | Type | Default |
|---|---|---|
| `popover` | `Popover` | — required |

### PopoverContent

| Prop | Type | Default |
|---|---|---|
| `popover` | `Popover` | — required |
| `aria-label` | `string` | — |
| `aria-labelledby` | `string` | — |
