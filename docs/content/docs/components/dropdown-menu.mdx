## Usage

```tsx
import { buttonVariants } from "@/components/ui/button";
import {
  DropdownMenu,
  DropdownMenuContent,
  DropdownMenuItem,
  DropdownMenuLabel,
  DropdownMenuSeparator,
  DropdownMenuTrigger,
} from "@/components/ui/dropdown-menu";

@Component()
class AccountMenu extends StatefulComponent {
  @State() dropdown = new DropdownMenu();

  render() {
    return (
      <>
        <DropdownMenuTrigger dropdown={this.dropdown} class={buttonVariants({ variant: "outline" })}>
          Open menu
        </DropdownMenuTrigger>
        <DropdownMenuContent dropdown={this.dropdown} aria-label="Account menu">
          <DropdownMenuLabel>My Account</DropdownMenuLabel>
          <DropdownMenuSeparator />
          <DropdownMenuItem dropdown={this.dropdown} value="profile" onSelect={() => {}}>
            Profile
          </DropdownMenuItem>
          <DropdownMenuItem dropdown={this.dropdown} value="logout" variant="destructive" onSelect={() => {}}>
            Log out
          </DropdownMenuItem>
        </DropdownMenuContent>
      </>
    );
  }
}
```

`DropdownMenu`/`DropdownMenuTrigger` are Morphos's `Dropdown`/`DropdownTrigger`, re-exported and
renamed to match shadcn/ui's naming.
