## Usage

```tsx
import { ContextMenu, ContextMenuContent, ContextMenuItem, ContextMenuTrigger } from "@/components/ui/context-menu";

@Component()
class FileRow extends StatefulComponent {
  @State() menu = new ContextMenu();

  render() {
    return (
      <>
        <ContextMenuTrigger contextMenu={this.menu} class="rounded-md border border-dashed p-8 text-center text-sm">
          Right-click here
        </ContextMenuTrigger>
        <ContextMenuContent contextMenu={this.menu} aria-label="File actions">
          <ContextMenuItem contextMenu={this.menu} value="rename" onSelect={() => {}}>Rename</ContextMenuItem>
          <ContextMenuItem contextMenu={this.menu} value="delete" variant="destructive" onSelect={() => {}}>
            Delete
          </ContextMenuItem>
        </ContextMenuContent>
      </>
    );
  }
}
```

## Props

### ContextMenu

Re-exported directly from `@morphos/overlays` — instantiate it in state (`@State() menu = new
ContextMenu()`) and pass it to every part below.

| Prop | Type | Default |
|---|---|---|
| `onOpenChange` | `(open: boolean) => void` | — |

### ContextMenuTrigger

| Prop | Type | Default |
|---|---|---|
| `contextMenu` | `ContextMenu` | — required |

### ContextMenuContent

| Prop | Type | Default |
|---|---|---|
| `contextMenu` | `ContextMenu` | — required |
| `aria-label` | `string` | — |

### ContextMenuItem

| Prop | Type | Default |
|---|---|---|
| `contextMenu` | `ContextMenu` | — required |
| `value` | `string` | — required |
| `label` | `string` | — |
| `disabled` | `boolean` | — |
| `onSelect` | `() => void` | — |
| `variant` | `"default" \| "destructive"` | `"default"` |
| `inset` | `boolean` | — |

`ContextMenuSeparator` takes no props beyond `class`.
