## Usage

```tsx
import {
  NavigationMenu,
  NavigationMenuContent,
  NavigationMenuItem,
  NavigationMenuLink,
  NavigationMenuList,
  NavigationMenuTrigger,
} from "@/components/ui/navigation-menu";

@Component()
class SiteNav extends StatefulComponent {
  @State() nav = new NavigationMenu();
  @State() productsItem = new NavigationMenuItem({ nav: this.nav, value: "products" });

  render() {
    return (
      <NavigationMenu aria-label="Main navigation">
        <NavigationMenuList nav={this.nav}>
          <NavigationMenuItem nav={this.nav} value="products">
            <NavigationMenuTrigger item={this.productsItem}>Products</NavigationMenuTrigger>
            <NavigationMenuContent item={this.productsItem}>
              <NavigationMenuLink href="/analytics">Analytics</NavigationMenuLink>
              <NavigationMenuLink href="/automation">Automation</NavigationMenuLink>
            </NavigationMenuContent>
          </NavigationMenuItem>
        </NavigationMenuList>
      </NavigationMenu>
    );
  }
}
```

<Callout type="info">
  Each `NavigationMenuItem` needs its own sibling instance (`this.productsItem`) — one per
  submenu — the same "instantiate once, pass to every part" pattern as the top-level `nav`.
</Callout>

## Props

### NavigationMenu

| Prop | Type | Default |
|---|---|---|
| `orientation` | `"horizontal" \| "vertical"` | — |
| `aria-label` | `string` | — |

### NavigationMenuList

| Prop | Type | Default |
|---|---|---|
| `nav` | `NavigationMenu` | — required |

### NavigationMenuItem

| Prop | Type | Default |
|---|---|---|
| `nav` | `NavigationMenu` | — required |
| `value` | `string` | — required |

### NavigationMenuTrigger

| Prop | Type | Default |
|---|---|---|
| `item` | `NavigationMenuItem` | — required |

### NavigationMenuContent

| Prop | Type | Default |
|---|---|---|
| `item` | `NavigationMenuItem` | — required |

### NavigationMenuLink

| Prop | Type | Default |
|---|---|---|
| `href` | `string` | — |
| `target` | `string` | — |
| `rel` | `string` | — |
| `onClick` | `(event: MouseEvent) => void` | — |
