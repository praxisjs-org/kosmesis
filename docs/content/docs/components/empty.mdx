## Usage

```tsx
import { Button } from "@/components/ui/button";
import {
  Empty,
  EmptyContent,
  EmptyDescription,
  EmptyHeader,
  EmptyMedia,
  EmptyTitle,
} from "@/components/ui/empty";

<Empty>
  <EmptyHeader>
    <EmptyMedia variant="icon">📭</EmptyMedia>
    <EmptyTitle>No messages yet</EmptyTitle>
    <EmptyDescription>When you receive a message, it'll show up here.</EmptyDescription>
  </EmptyHeader>
  <EmptyContent>
    <Button>Compose a message</Button>
  </EmptyContent>
</Empty>
```
